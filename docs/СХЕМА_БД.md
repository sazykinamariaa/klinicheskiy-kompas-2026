# Схема базы данных «Клинического компаса»

База — SQLite (файл в папке data/, путь задаёт KK_DATA_DIR), доступ через SQLAlchemy. Таблицы создаются при первом запуске; начальные данные — backend/app/seed.py (KK_SEED=1).
Персональные поля (комментарии, заметки) шифруются при хранении ключом data/secret.key или KK_SECRET_KEY.

## adherence_events
id, patient_id →patients.id, day, kind, reason

## ae_reports — нежелательные явления
id, patient_id →patients.id, severity, urgent_flag, persists, worsening, details, created_at

## alert_events — история алерта: принят, эскалирован, закрыт
id, alert_id →alerts.id, ts, kind, actor_id →users.id, data

## alerts — алерты (сигналы)
id, patient_id →patients.id, rule_id, level, title, status, basis, facts, created_at, accepted_at, accepted_by →users.id, closed_at, close_data, after_hours, patient_notice, open_key

## audit_log — журнал действий: кто, что, когда
id, ts, actor_id →users.id, actor_role, action, entity, entity_id, patient_id, detail

## checkins — ежедневные отметки
id, patient_id →patients.id, day, answers, complete, created_at

## clinician_flags
id, patient_id →patients.id, flag, value, set_by →users.id, at

## consents
id, patient_id →patients.id, version, text_status, accepted_at, revoked_at

## doctor_marks
id, patient_id →patients.id, kind, by →users.id, at

## goals
id, patient_id →patients.id, text, set_by →users.id, at

## import_jobs
id, patient_id →patients.id, created_by →users.id, created_at, source_text, extracted, status, reviewed_by →users.id

## intake_answers — ответы первичной анкеты и уточнения
id, patient_id →patients.id, question_id, answer, updated_at

## invitations — приглашения пациентов
id, patient_id →patients.id, code, expires_at, accepted_at

## med_items — принимаемые препараты
id, patient_id →patients.id, name, dose, source, active, confirmed_by_doctor

## messages — сообщения врач — пациент
id, patient_id →patients.id, sender_id →users.id, body, kind, ts, read_at

## my_plans
id, patient_id →patients.id, sections, version, edited_by →users.id, edited_at, confirmed_by_doctor →users.id, confirmed_at

## notification_log
id, outbox_id →outbox.id, alert_id →alerts.id, channel, recipient, status, ts, detail

## outbox
id, kind, payload, idempotency_key, status, attempts, next_attempt_at, last_error, alert_id →alerts.id, created_at, sent_at

## patients — пациенты
id, code, sex, birth_date, doctor_id →users.id, user_id →users.id, status, checkin_enabled_on, synthetic, created_at

## profile_field_values — клинические факты врача (диагноз, аллергии и др.)
id, patient_id →patients.id, field_id, value, entered_by →users.id, entered_at

## proposals — предложения системы и решения врача
id, patient_id →patients.id, kind, key, payload, basis, status, created_at, decided_by →users.id, decided_at, reason, decision_data, open_key

## psy_decisions — решения психиатра по рекомендациям
id, patient_id →patients.id, item_key, decision, comment, by →users.id, at

## psych_notes — заключения психиатра
id, referral_id →referrals.id, author_id →users.id, decision, comment, ts

## referral_events — статусы направления
id, referral_id →referrals.id, kind, actor_id →users.id, at, note

## referrals — направления к психиатру
id, patient_id →patients.id, rule_id, kind, tier, due_at, status, proposal_id →proposals.id, confirmed_by →users.id, confirmed_at, summary, deferred_until

## safety_followups
id, patient_id →patients.id, answers, complete, sf6_days_ago, started_at, completed_at

## scale_runs — заполнения PHQ-9 и GAD-7
id, patient_id →patients.id, scale_id, answers, complete, total, started_at, completed_at

## shift_handovers
id, author_id →users.id, created_at, data, next_responsible_id →users.id, confirmed_by →users.id, confirmed_at

## status_changes
id, patient_id →patients.id, status, proposal_id →proposals.id, confirmed_by →users.id, confirmed_at

## symptom_entries
id, patient_id →patients.id, name, present, duration_weeks, entered_by →users.id, at

## therapy_orders — назначения препарата
id, patient_id →patients.id, drug_id, regimen, proposal_id →proposals.id, confirmed_by →users.id, confirmed_at, started_on, stopped_at, off_label

## tokens
id, user_id →users.id, token_hash, expires_at

## trusted_contacts
id, patient_id →patients.id, name, phone, consent

## users — пользователи и роли (пароль — хеш PBKDF2)
id, login, password_hash, role, name, phone, active, duty_role, created_at

## visits
id, patient_id →patients.id, at, note, created_by →users.id, created_at
